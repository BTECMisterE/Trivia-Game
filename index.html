function setupBuzzerListener() { rRef.child('currentBuzz').on('value', s => { const v = s.val(); const notif = document.getElementById('buzzer-notif'); const userSpan = document.getElementById('th-user'); if (v && v.user) { notif.style.display = 'block'; // We search for the player in the database to get their assigned team rRef.child('players').once('value', playersSnap => { let teamNum = "??"; playersSnap.forEach(p => { if (p.val().name === v.user) { teamNum = p.val().team || "??"; } }); // Display both Name and Team Number userSpan.innerHTML = `${v.user}
TEAM ${teamNum}`; }); } else { notif.style.display = 'none'; } }); }
